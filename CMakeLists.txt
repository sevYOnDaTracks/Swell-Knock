cmake_minimum_required(VERSION 3.22)
project(SwellKnock VERSION 0.1.0 LANGUAGES C CXX)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(CMAKE_CXX_STANDARD 17)
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.6 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
set(KNOCK_FORMATS VST3 Standalone)
if(APPLE)
  list(APPEND KNOCK_FORMATS AU)
endif()
juce_add_plugin(SwellKnock COMPANY_NAME "Swell-audio" BUNDLE_ID "fr.swell-audio.knock" PLUGIN_MANUFACTURER_CODE Swel PLUGIN_CODE Sknk FORMATS ${KNOCK_FORMATS} PRODUCT_NAME "Swell Knock" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
juce_add_binary_data(KnockAssets SOURCES swell-mark.png)
target_sources(SwellKnock PRIVATE Source/Plugin.cpp)
target_compile_definitions(SwellKnock PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(SwellKnock PRIVATE KnockAssets juce::juce_audio_utils juce::juce_dsp PUBLIC juce::juce_recommended_config_flags)
enable_testing()
add_executable(KnockDSPTest Tests/DSPTest.cpp)
add_test(NAME KnockDSP COMMAND KnockDSPTest)
option(KNOCK_BUILD_DEMO "Build offline VST3 demo renderer" OFF)
if(KNOCK_BUILD_DEMO)
  juce_add_console_app(KnockDemo PRODUCT_NAME "Knock Demo Renderer")
  target_sources(KnockDemo PRIVATE Tools/RenderDemo.cpp)
  target_compile_definitions(KnockDemo PRIVATE JUCE_PLUGINHOST_VST3=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
  target_link_libraries(KnockDemo PRIVATE juce::juce_audio_utils)
endif()
